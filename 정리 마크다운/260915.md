### 변수와 연산자

#### var, let, const

- 변수 : 값을 담는 공간
  - 변수명 = 값. 변수를 지정하는 것 -> "변수를 선언하다"라고 한다.
  - var, let, const의 차이? -> 면접질문 多

```html
<script>
  // var = 예전에 사용. 값을 선언하기 전에 출력 요청해도 undefined로 출력됨.
  // 예전에 사용했던 문법이라 현재도 존재는 하나, 지금으로서는 사용하는 것 비추천.
  var oldVar = 1;
  console.log(oldVar);
  oldVar = 2;
  console.log(oldVar);

  // let = 값을 바꿀 수 있는 변수에 사용 (가끔 사용)
  let nickname = "cool";
  console.log(nickname);
  nickname = "개발자 3호";
  console.log(nickname);

  // const = 값을 바꿀 수 없는 변수에 사용 (자주 사용)
  const name = "성시원"; // ->
  console.log(name);
  name = "성*원";
</script>
```

#### 자료형과 typeof

반드시 문자열(string)은 "" 또는 '' 안에, 숫자(number)는 그냥 작성.

```html
<script>
  const userId = "개발자";
  console.log(typeof userId); // 변수의 타입을 알고 싶을 땐 typeof 를 변수명 앞에 붙인다.
  const age = 24;
  console.log(typeof age);
  check = typeof true; // true는 문자가 아닌 논리형(boolean ~ 불리언)이며, 참 or 거짓을 의미한다.
  check = false;
  console.log(check);

  let unknown; // unknown은 undifined(정의되지 않음)가 타입이다. -> 값이 정의되지 않았음을 의미.
  console.log(typeof unknown);

  let isNone = null; // null은 값이 없음을 의미하며, object(객체) 타입이다.
  console.log(typeof isNone);
</script>
```

#### 숫자

```html
<script>
  const num1 = 10;
  const num2 = 3;
  console.log(num1 + num2); // 10 + 3 = 13
  console.log(num1 - num2); // 10 - 3 = 7
  console.log(num1 * num2); // 10 * 3 = 30

  console.log(num1 / num2); // 10 / 3 = 3.3333333333333335
  console.log(num1 % num2); // 10 / 3의 나머지 = 1
  console.log(parseInt(num1 / num2)); // 10 / 3에서의 정수(소수점 제거) = 3

  // 문자 to 숫자
  console.log(Number("100")); // Number(값)
  console.log(Number("Hello")); // NaN (Not a Number) -> 숫자가 아닌 값.

  // 숫자 to 문자
  console.log(String(100)); // String(숫자)

  // 실수로 변환
  console.log(typeof parseFloat("3.14")); // parseFloat(문자)
</script>
```

#### Boolean과 비교 연산자

```html
<script>
  console.log(true, false);

  // 모든 값을 Boolean으로 변환 -> Boolean(값)
  // false인 경우 : 0, ''(빈 문자열), undefined, null, NaN / 그 外 전부 true.
  console.log(Boolean("Hello")); // true
  console.log(Boolean("")); // false

  console.log(Boolean(0)); // false
  console.log(Boolean(1)); // true

  const a = 100;
  const b = "100";
  // ==와 ===의 차이는? : 둘 다 양쪽이 같음을 묻는 질문이다. ==은 값만을 비교, ===은 타입까지 비교.
  console.log(a == b); // true
  console.log(a === b); // false

  // !를 앞에 붙이면 부정을 표시한다. ∴ 양쪽의 값이 같지 않은가?와 양쪽의 값과 타입이 둘 다 같지 않은가?를 비교.
  console.log(a != b); // false
  console.log(a !== b); //true

  // 비교 연산자인 <, >, <=, >=는 무조건 결과값이 Boolean으로만 나온다.
  const c = 200;
  console.log(a < c); // 100 < 200 = true
  console.log(a > c); // 100 > 200 = false
  console.log(a <= c); // 100보다 200이 크거나 같은가(이상)? true
  console.log(a >= c); // 100이 200보다 작거나 같은거(이하)? false
</script>
```

#### 논리 연산자

```html
<script>
  const x = 10;
  const y = 20;

  && : 그리고(and) -> 무조건 둘 다 true여야 true 출력.
  console.log(x < 15 && y > 10); // x(10) < 15 그리고 y(20) > 10 -> true
  console.log(x > 15 && y > 10); // x(10) > 15 그리고 y(20) > 10 -> false

  console.log("JavaScript" && 12345); // 12345 출력 -> true값과 true값이 같이 있는데, 비교연산자가 없으므로 12345만 출력.
  console.log("" && 12345); // 공란("") 출력. ∵ 앞의 결과값이 false이므로 거기에서 계산을 멈춰버림.

  || : 또는(or) -> 둘 중 하나만이라도 true일 경우 true 출력.
  console.log(x < 15 || y > 10); // x(10) < 15 또는 y(20) > 10 -> true
  console.log(x > 15 || y > 10); // x(10) > 15 또는 y(20) > 10 -> true

  console.log(null || "JS"); // JS 출력. ∵ 앞의 결과값이 false이므로 true를 찾을 때까지 계산을 멈추지 않고 이어감.
  console.log(1 || "JS"); // 1 출력. ∵ 앞의 결과값이 true이므로 더 이상 값을 찾지 않고 계산을 멈춰버림.
  // --> 단축 평가 (Short circuit evaluation) : 앞의 값에 따라서 뒤의 값을 평가할지 여부를 결정하는 것을 의미한다.
  // --> 로그인 상태 확인, 입력값 검증 등의 상황에 활용한다.
</script>
```

### 조건문

#### if / else

if와 else의 형식 : if(조건) {조건이 true인 경우} else {조건이 false인 경우}

```html
<script>
  let age = 19;
  // 나이가 19세보다 많으면 "성인", 아니면 "미성년자" -> if 조건이
  true일 경우 else 부분은 보지 않고 계산을 마친다.
  if (age > 19) {
  console.log("성인"); } else { console.log("미성년자"); } if (age > 19) {
  console.log("성인"); } else { console.log("미성년자"); }

  삼항 연산자 : 조건 ? 조건이 true : 조건이 false
  const result = age > 19 ? "성인" : "미성년자";
  // 조건에 대한 결과값이 result에 담기므로 result 출력 필요.
        console.log(result);
</script>
```

#### else if

```html
<script>
  /* 형식
        if(조건A) { 조건A가 true
        else if (조건B) { 조건A가 false, 조건B가 true }
        else if(조건C) { 조건A, 조건B가 false, 조건C가 true }
        else { 모든 조건이 false일 경우 }
          -> if만 써도 가능, else 생략 가능, else if는 여러 개 사용 가능.
      */

  ㅁ 20세 이상이면 성인, 14세 이상이면 청소년, 8세 이상이면 어린이, 그 외에는 유아
  age = prompt("나이를 입력해주세요"); // 알림창이 나오며 입력해달라는 창이 나온다.
  if (age >= 20) {
    console.log("성인");
  } else if (age >= 14) {
    console.log("청소년");
  } else if (age >= 8) {
    console.log("어린이");
  } else {
    console.log("유아");
  }
</script>
```

#### switch

```html
<script>
  const number = 7;
  // number값이 짝수이면 "짝수", 홀수이면 "홀수", 그게 아니면 "숫자가 아니다" 출력 -> 나머지 연산자(%)를 사용한다.
  // !===는 없는 문법이다. !==가 타입을 비교하면서 값도 맞는지 비교하는 문법이다.
  // 짝수는 2로 나누었을 때 나머지가 0, 홀수는 2로 나누었을 때 나머지가 1 (나머지가 0이 아님)
  console.log(number % 2 === 0);

  if (number % 2 === 0) {
    console.log("짝수");
  } else if (number % 2 === 1) {
    console.log("홀수");
  } else {
    console.log("숫자가 아니다");
  }

  /*
          switch(데이터) {
          case 조건A(값):
              조건A가 true
              break;
          case 조건B(값):
              조건B가 true
              break;
          case default:
              모든 조건이 false
          }
         */

  // switch문에서 break를 걸지 않았을 경우, 본인의 조건이 아닌 틀린 값임에도 이후에 값을 출력하고 나간다.
  // break가 내보내는 역할이므로 마지막에는 break로 내보내지 않아도 된다.
  switch (number % 2) {
    case 0:
      console.log("짝수");
      break;
    case 1:
      console.log("홀수");
      break;
    default:
      console.log("숫자가 아니다");
  }
</script>
```
