## 객체

객체 생성과 속성/함수 추가 <br>
빈 객체에 속성이랑 함수 추가

```js
/*
    사람 {
    // 특징들 => 변수
        이름, 나이, 사는 곳,
    // 행동들 => 함수(기능)
        생활패턴(기상/식사/씻기/착의/출발)
    }
*/

// 변수끼리 서로 관련이 있다는 게 보이지 X.
const name = "성현제";
const age = 10;
const addr = "서울";

// 배열은 서로 관련이 있다는 것으로 보이나, 각 값들이 무얼 의미하는지 알 수 X.
const personArr = ["성현제", 10, "서울"];

// 객체
const person = {
  // 변수 (변수명 : 값)
  name: "성현제",
  age: 10,
  addr: "서울",
  // 함수
  hello() {
    // this : 본인 자체 -> {this.변수명}
    console.log(`안녕하세요, ${this.name}입니다.`);
  },
};
console.log(person); // 모든 변수와 함수를 출력한다.
console.log(person.name); // person 중 name값만 단독으로 출력한다.
person.hello(); // 변수명.함수명 => 해당 함수의 값이 출력된다.

// 빈 객체와 속성/함수 추가
const person1 = {}; // const person1 = new Object();
person1.name = "홍길동"; // 객체 추가 또는 수정법 1
person1["age"] = 9; // 객체 추가 또는 수정법 2
person1.hello = function () {
  console.log(`나는 ${this.name}이고, 나이는 ${this.age}살이다.`);
};
person1.hello();

// 일반 함수와 화살표 함수의 this 차이는?
const person2 = {
  name: "박하윤",
  hello: function () {
    // 익명함수는 hello(){}로 축약 가능하다.
    console.log(`익명 함수 : ${this.name}`);
  },
};
const person3 = {
  name: "김수영",
  // 객체 함수로 화살표 함수 비권장
  hello: () => {
    console.log(this); // 전역 객체인 Window를 가리킨다.
    console.log(`화살표 함수 : ${this.name}`);
  },
};

person2.hello(); // 정상적으로 출력
person3.hello(); // 이름 제외하고 출력

// 객체도 구조분해할당이 가능하다. but 기존 변수명 사용 불가
const { name: name1, age: age1 } = person1;
console.log(name1, age1);
```

```js
// 생성자 함수, 클래스 -> 변수와 함수는 첫글자를 소문자, 클래스는 첫글자를 대문자로 표기
function Person(name, age) {
  this.name = name;
  this.age = age;
  this.hello = function () {
    console.log(`안녕? ${this.name}야. 나이는 ${this.age}살이야.`);
  };
}

const p1 = new Person("김지아", 5);
p1.hello();

// 함수 형식보다는 클래스 형식 사용 多
class Person2 {
  // constructor는 객체 생성 시 바로 호출된다.
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  hello() {
    console.log(`안녕? ${this.name}야. 나이는 ${this.age}살이야.`);
  }
}

const p2 = new Person2("김수아", 2);
p2.hello();
```

## 내장 객체

Set, Math, Date, JSON

```js
Set 중복 제거용
const arr = [1, 2, 2, 3, 3, 4, 4, 4, 5];
console.log(arr);
const unique = new Set(arr);
console.log(unique);
console.log([...unique]); // Set으로 바뀐 속성을 다시 배열로 바꾼다.

Math 계산용 (수학적인 것들)
console.log(Math.max(5, 7, -1, -8)); // 최대값
console.log(Math.min(5, 7, -1, -8)); // 최소값
console.log(Math.round(2.897)); // 3 -> 반올림
console.log(Math.floor(2.897)); // 2 -> 내림
console.log(Math.ceil(2.897)); // 3 -> 올림
console.log(Math.random()); // 0 ~ 1까지의 랜덤 (소수점 포함) -> 범위는 0 <= Math.random < 1

// 1 ~ 10까지의 랜덤 숫자가 필요할 경우?
// 1 ~ 10까지의 랜덤 숫자
// 0 0 <= Math.random() < 1
// * 10
// 0 <= Math.random() * 10 < 10
console.log(Math.random() * 10);
// Math.floor()
console.log(Math.floor(Math.random() * 10));
// +1
console.log(Math.floor(Math.random() * 10) + 1);

// 5 ~ 15까지의 랜덤 숫자가 필요할 경우
// 5 <= Math < 16
console.log(Math.floor(Math.random() * 11) + 5); // 숫자가 필요한 개수만큼 곱한다.

Date 객체
const now = new Date();
console.log(now); // 현재 날짜, 시각, 표준시까지 표기된다.
// getMonth : 0 ~ 11월. 1월을 0으로 표기한다.
console.log(
  `${now.getFullYear()}년 ${now.getMonth() + 1}월 ${now.getDate()}일`,
);
// const date = new Date(2027, 0, 28, 14, 20, 0); -> 특정 날짜가 필요할 때에도 월만 1을 차감해줘야 한다.

const date = new Date("2027-01-28T14:20:00"); // 문자열로 작성 시 제대로 출력. ("년도4자리-월2자리-일자2자리-T시-분-초")
console.log(date);
console.log(date.toLocaleDateString("ko-KR")); // 한국 표준시로 변경하는 방법. 보통 위처럼 값을 불러온다.

Timer
// 밀리세컨드 기준, 1000 = 1초
// setTimeout 일정 시간 후 1회 실행
setTimeout(() => {
  console.log("3초 후 실행!");
}, 3000);

// setInterval 일정 시간마다 반복 실행 (기본적으로 종료 X)
let sec = 0;
setInterval(() => {
  console.log(`${++sec}초!`);
}, 1000); // 지정한 간격마다 발생

JSON(JavaScript Object Notation)
// 서버에서 클라이언트로 JSON 객체로 통신 (localStorage와 관련)
// 객체 -> 문자열
const jsonStr = JSON.stringify(person);
console.log(typeof jsonStr); // string
// 문자열 -> 객체
const jsonObj = JSON.parse(jsonStr);
console.log(typeof jsonObj); // object
```
