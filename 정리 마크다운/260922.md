## 문서 객체 가져오기

```html
<script src="./07-dom.js"></script>
```

```html
<div id="testId" class="testClass" name="testName">Hello, World!</div>
<div id="testId2" class="testClass" name="testName">Hello, JavaScript!</div>
```

```js
// HTML과 DOM의 차이?
// html은 화면에 보여줄 동적인 텍스트 / DOM은 html을 해석해 만든 객체 구조이며, 동적으로 변경될 수 있다.

// 문서 객체 가져오기
console.log(document.body);

// 태그로 가져오기
console.log(document.getElementsByTagName("h1"));

// class로 가져오기
console.log(document.getElementsByClassName("testClass"));

// name으로 가져오기
console.log(document.getElementsByName("testName"));

// id로 가져오기 -> 맨 위의 아이디 내용 하나밖에 못 가져오므로 태그에 s가 없다.
console.log(document.getElementById("testId"));

// querySelector(선택자)
console.log(document.querySelector("div")); // -> 무조건 한 개의 구문만 출력
// querySelectorAll(선택자)
console.log(document.querySelectorAll("div")); // -> 여러 개의 배열을 출력
이 두가지만 알아도 괜찮다.
```

### div 색상 변경

### 문서 객체 추가/삭제

```html
<div id="testId3"></div>
```

```js
문서 객체 추가
const testId3 = document.querySelector("#testId3");
// = testId3.innerHtml = "<p>텍스트 추가</p>"
const p = document.createElement("p"); // p 태그 생성
p.textContent = "텍스트 추가"; // <p>텍스트 추가</p>
testId3.appendChild(p);

const pTarget = document.querySelector("#testId3 p"); // #testId3 안의 p태그를 선택한다.
pTarget.remove();
```

### 문자열 분리 ul 출력

```html
<label id="la">사과, 바나나, 오렌지, 포도</label>
<div id="result5"></div>
<button onclick="stringSplit()">출력</button>
```

```js
const la = document.querySelector("#la");
const result5 = document.querySelector("#result5");
const stringSplit = () => {
  const arr = la.textContent.split(", ");
  const ul = document.createElement("ul");
  for (const value of arr) {
    const li = document.createElement("li");
    li.textContent = value.trim(); // trim 앞뒤 공백 제거
    ul.appendChild(li);
  }
  result5.appendChild(ul);
};
```

### li 요소 추가 & 삭제

```html
<ul id="list"></ul>
<button onclick="addItem()">아이템 추가</button>
<button onclick="removeItem()">아이템 삭제</button>
```

```js
const list = document.querySelector("#list");
const addItem = () => {
  const li = document.createElement("li");
  li.textContent = "아이템 추가";
  list.appendChild(li);
};
const removeItem = () => {
  const item = document.querySelector("#list li");
  item.remove();
};
```
