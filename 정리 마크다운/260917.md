## 반복문

```html
<script>
  // 1~10까지 출력 할 경우 : for문 -> for(초기값; 조건식-범위; 증감식;) {반복문 조건에 해당되는 만큼 실행}
  for (let i = 1; i <= 10; i++) {
    console.log(i);
  }
</script>
```

## 전위/후위 연산자

```html
<script>
  let num = 1;
  num = num + 1;
  console.log(num); // 2
  // num = num + 1 : ++num, num++ --> 변수명 앞에 ++를 붙이면 1만큼 값이 올라간다.
  // num = num - 1 : --num, num-- --> 변수명 뒤에 --를 붙이면 1만큼 값이 내려간다.
  // ++num, --num 전위 연산자 : 먼저 계산하고 나중에 출력
  // num++, num-- 후위 연산자 : 먼저 출력하고 나중에 계산
  console.log(++num); // 3
  console.log(num++); // 결과값은 3이나, 실제로 num은 4가 되었다.
  console.log(num); // 4
  console.log(--num); // 3
  console.log(num--); // 결과값은 3이나, 실제로 num은 2가 되었다.
</script>
```

## 복합대입연산자

```html
<script>
  num += 3; // num = num + 3과 같은 의미
  num -= 2; // num = num - 2와 같은 의미
  num *= 5; // num = num * 5와 같은 의미
  num /= 4; // num = num / 4와 같은 의미

  // 30부터 3씩 감소해서 3까지 출력하려면?
  for (let i = 30; i >= 3; i -= 3) {
    console.log(i);
  }
</script>
```

## 배열과 for문

```html
<script>
  const movies = [
    "오디세이",
    "옵세션",
    "스파이더맨: 브랜드 뉴 데이",
    "더 드라마",
    "싱 어게인",
  ];
  // i <= 4는 i < 5와 같음
  // 배열.length는 배열의 길이
  for (let i = 0; i < movies.length; i++) {
    console.log(movies[i]);
  }

  // for .. in : for(인덱스 in 배열) {}
  for (let index in movies) {
    console.log(movies[index]);
  }

  // for .. of : for(값 of 배열) {}
  for (const value of movies) {
    console.log(value);
  }
</script>
```

## while문

```html
<script>
  /* while(조건) {조건이 true이면 계속} */

  // 1 ~ 10까지 출력
  let i = 1;
  while (i <= 10) {
    console.log(i);
    i++;
  }

  //무한루프 - 일부러 사용 (eg.게임)
  while (true) {
    const input = prompt("종료하려면 q를 입력해주세요");
    //break를 만나면 해당 구문을 빠져나가는 조건 -> 반드시 필요!
    if (input === "q") break;
  }
</script>
```

## break / continue

```html
<script>
  // break와 continue의 차이는?
  // break : 반복문을 완전히 빠져나간다.
  // continue : 해당 조건만 실행하지 않고 건너뛰어 반복문을 다시 실행하러 앞으로 돌아간다.
  for (let i = 1; i < 10; i++) {
    if (i === 7) break;
    if (i === 2) continue;
    console.log(i); // 1, 3, 4, 5, 6
  }
</script>
```

## 2차원 배열 + 중첩 for문

```html
<script>
  const arr = [
    [1, 2, 3],
    [4, 5],
    [6, 7, 8, 9],
  ];
  // 반복문을 사용하여 1부터 9까지 출력하려면?
  console.log(arr[0]); // 1, 2, 3
  console.log(arr[0][0]); // -> 1. 앞은 배열의 인덱스값, 뒤는 그 배열 안의 인덱스값.
  console.log(arr[0][1]); // 2
  for (let i = 0; i < arr.length; i++) {
    // console.log(arr[i]);
    for (j = 0; j < arr[i].length; j++) {
      console.log(arr[i][j]);
    }
  }

  for (const i of arr) {
    for (const j of i) {
      console.log(j);
    }
  }
</script>
```
