```js
// 1. 매개변수 X, 리턴값 X
함수 정의의 형태 : function 함수명() {호출할 때 발생할 내용}
// 함수를 정의하고 그 안에 출력 태그를 적어두면, 함수를 호출할 때에 내부의 출력 태그에서 출력된다.
function sayHello() {
  console.log("안녕하세요!");
}
sayHello(); // 함수 호출 -> 안녕하세요! 출력
sayHello();

// 2. 매개변수 O, 리턴값 X
function greet(name = "guest") {
name에 아무 값도 입력하지 않았을 경우 기본값 지정 방법 : 변수명 = 기본값 내용
  console.log(`${name}님, 안녕하세요!`);
}
greet("서연우"); // --> "서연우님, 안녕하세요!"" 출력
greet("하현성"); // --> "하현성님, 안녕하세요!"" 출력
greet(); // name에 아무 값도 지정하지 않고, 기본값도 없을 경우 --> undefined 출력

// 3. 매개변수 X, 리턴값 O
return : 함수 호출 결과, 함수 종료; 2가지 의미를 가진다.
function getNumber() {
  return 1049;
}
getNumber(); // 아무 일도 일어나지 않는다.
console.log(getNumber()); // --> 1049 출력

// 4. 매개변수 O, 리턴값 O
function add(a, b) {
  return a + b;
}
console.log(add(10, 20)); // 두 가지 이상의 값을 보낼 때 ,로 구분 --> 30 출력
console.log(add(30, 50)); // 80 출력 : 필요한 값만 넣으면 결과가 출력된다.
```

## 변수의 유효범위(Scope)

전역 변수와 지역 변수의 차이

```js
const num1 = 100; // 전역변수 -> 함수 내외 둘 다 사용 가능하다.
function sample1() {
  const num1 = 200; // 지역변수(=지역 스코프) -> 함수 내부에서만 사용 가능하다.
  console.log(`sample1 내부 : ${num1}`); // 200 출력 -> 바로 윗줄의 num1 주석 걸 경우 100으로 출력.
}
sample1();
console.log(`sample1 외부 : ${num1}`); // --> 100 출력
```

## 선언적 vs 익명 vs 화살표 함수

함수 선언 방식 비교

```js
// 선언적 함수 (Function Declaration)
★ 함수를 만들기 전에 함수를 호출해도 결과값이 출력된다. (= 호이스팅) -> 함수를 어디에 만들었는지 찾기 힘들어진다.
★ 익명 함수와 화살표 함수 = 함수를 만들기 전에 함수를 호출하면 에러가 난다.
function multiply(a, b) {
  return a * b;
}
console.log(multiply(3, 4)); // --> 12 출력

// 익명 함수(Function Expression) : 함수명이 없다. ∴ 변수명 선언 = function () {}
const subtract = function (a, b) {
  return a - b;
};
console.log(subtract(5, 8));

// 화살표 함수(Arrow Function) : function 대신 => 사용
const divide = (a, b) => {
  return a / b;
  // (a, b) => a / b; 로 생략 가능.
};
console.log(divide(12, 3));
```

## 콜백 함수

함수를 파라미터로 전달

```js
function callFunc(callback) {
  console.log("함수 호출 전~");
  callback();
  console.log("함수 호출 후!");
}
function call() {
  console.log("안녕하세요~ 콜백 함수 호출!");
}
callFunc(call);

함수 호출 전 -> 콜백 함수 호출 -> 함수 호출 후 순서로 출력.
```

## 배열 관련 콜백 함수들

forEach / map / filter / reduce

```js
const numbers = [1, 2, 3, 4, 5];

// for (let i = 0; i < numbers.length; i++) {console.log(numbers[i]);} 과 아래 함수의 값은 같다.
★ forEach 배열의 각 요소마다 콜백 실행. -> 형태 : 변수명.forEach((value, index, array) => {console.log(출력할 것)})
// value 값을 전개하여 출력 / index 인덱스값 전개하여 출력 / array 배열을 값 개수만큼 반복하여 출력
numbers.forEach((value, index, array) => {
  console.log(value);
});

★ map 각 요소를 콜백으로 변형해서 "새 배열"을 만들어 반환한다.
// numbers.map((num) => {console.log(num)});
const doubled = numbers.map((num) => {
  return num * 2;
});
console.log(doubled); // --> [2, 4, 6, 8, 10] 출력 / console.table(변수명) -> 인덱스와 값이 표로 출력된다.

★ filter 조건이 걸렸을 때. : 콜백이 true를 반환하는 요소만 골라서 "새 배열"을 만들어 반환한다.
const even = numbers.filter((value) => {
  return value % 2 === 0;
});
console.log(even); // [1, 2, 3, 4, 5] 中 return 조건이 true인 값들 [2, 4]만 출력

★ reduce 배열의 각 값들을 누적하여 하나의 값으로 반환한다.
// accumulator(누적된 값), value, index, array
const sum = numbers.reduce((acc, value) => {
  return acc + value;
}, 0); // acc = 0으로 초기값 설정.
console.log(sum); // 1 + 2 + 3 + 4 + 5 = 15 출력
```
