### DML 중 하나이기도 하면서, ,DQL(Data Query Language) : 데이터 조회
\* : 모든 column들<br>
select * from cars;

- 내가 필요한 column만 <br>
select name, plate, mileage from cars;
- select A as B : A를 B로 별칭을 부여한다. (실제 column 이름이 바뀌는 것은 X)<br>
select name as 차종, mileage as 주행거리 from cars;

- 계산식도 가능하다.
select item, cost, cost * 1.1 as 부가세포함 from intenance_records;
- 겹치는 값은 한 번만<br>
select distinct fuel from cars;

where : 조건에 맞는 행만
- mileage가 150000 이상인 경우 - 주행거리가 150000이 넘는 차는?<br>
select name, mileage from cars where mileage >= 150000;

- car_type이 'SUV'이면서 fuel가 '가솔린'<br>
select * from cars where car_type = 'SUV' and fuel = '가솔린';
- fuel가 '전기'이거나 '하이브리드'<br>
select * from cars where fuel = '전기' or fuel = '하이브리드';
- in 안에 있는 조건 중 하나라도 있을 경우<br>
select * from cars where fuel in ('전기', '하이브리드');
- maintenance_records 테이블에서 cost가 100000 이상이면서 200000 이하<br>
select * from maintenance_records where cost >= 100000 and cost <= 200000;<br>
select * from maintenance_records where cost between 100000 and 200000;

- like + % : 0글자 이상 ---> 검색 기능 만들 때 사용<br>
select name, plate from cars where name like '%티%';

- null : =로 찾지 못한다. / is null 또는 is not null 사용<br>
- cars 테이블에서 production_year가 null인 경우 / null이 아닌 경우<br>
select * from cars where production_year is null; -- not은 부정.

- order by : 정렬 / 기본 asc는 오름차순, desc는 내림차순<br>
select name, mileage from cars order by mileage desc;<br>
select name, car_type, mileage from cars order by car_type desc,  mileage;

- limit 숫자 (한 페이지당 나오는 개수) offset 몇 번째 데이터부터 출력 ---> 페이징 처리<br>
select id, item, cost, service_date from maintenance_records limit 5;<br>
select id, item, cost, service_date from maintenance_records limit 5 offset 5;
